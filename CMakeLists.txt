CMAKE_MINIMUM_REQUIRED(VERSION 4.4)
PROJECT(lib3ds VERSION 2.0.0 LANGUAGES C CXX)

INCLUDE(GNUInstallDirs)
INCLUDE(CMakePackageConfigHelpers)

INCLUDE(CMakeDependentOption)

OPTION(BUILD_SHARED_LIBS "Build lib3ds as a shared library (OFF builds a static library)" ON)
OPTION(LIB3DS_BUILD_TOOLS "Build the example tools (3ds2obj, 3dsdump, cube, 3dsplay)" ON)
CMAKE_DEPENDENT_OPTION(LIB3DS_INSTALL_TOOLS "Install 3ds2obj and 3dsdump" ON
    "LIB3DS_BUILD_TOOLS" OFF)

SET(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${PROJECT_BINARY_DIR}/bin)
SET(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${PROJECT_BINARY_DIR}/bin)
SET(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${PROJECT_BINARY_DIR}/lib)

ADD_SUBDIRECTORY(src)
IF(LIB3DS_BUILD_TOOLS)
    ADD_SUBDIRECTORY(examples)
ENDIF(LIB3DS_BUILD_TOOLS)

# CMake package so consumers can use:
#   find_package(lib3ds CONFIG REQUIRED)
#   target_link_libraries(app PRIVATE lib3ds::lib3ds)
SET(LIB3DS_INSTALL_CMAKEDIR ${CMAKE_INSTALL_LIBDIR}/cmake/lib3ds)

INSTALL(EXPORT lib3dsTargets
    NAMESPACE lib3ds::
    DESTINATION ${LIB3DS_INSTALL_CMAKEDIR})

CONFIGURE_PACKAGE_CONFIG_FILE(
    ${PROJECT_SOURCE_DIR}/cmake/lib3dsConfig.cmake.in
    ${PROJECT_BINARY_DIR}/lib3dsConfig.cmake
    INSTALL_DESTINATION ${LIB3DS_INSTALL_CMAKEDIR})

WRITE_BASIC_PACKAGE_VERSION_FILE(
    ${PROJECT_BINARY_DIR}/lib3dsConfigVersion.cmake
    COMPATIBILITY SameMajorVersion)

INSTALL(FILES
    ${PROJECT_BINARY_DIR}/lib3dsConfig.cmake
    ${PROJECT_BINARY_DIR}/lib3dsConfigVersion.cmake
    DESTINATION ${LIB3DS_INSTALL_CMAKEDIR})
